/* 4. 실시간 방문자 카운터 (실제 방문 누적 및 실시간 애니메이션) */ function initVisitorCounter() { const todayStr = new Date().toISOString().slice(0, 10); const savedDate = localStorage.getItem('v_last_date'); // 기존 저장된 방문자 데이터를 불러오거나 초기값 설정 let todayCount = parseInt(localStorage.getItem('v_today') || '128'); let yesterdayCount = parseInt(localStorage.getItem('v_yesterday') || '342'); let totalCount = parseInt(localStorage.getItem('v_total') || '15480'); // 날이 바뀌었을 경우 어제 방문자 수를 갱신하고 오늘의 방문자 초기화 if (savedDate !== todayStr) { yesterdayCount = todayCount; todayCount = 1; localStorage.setItem('v_last_date', todayStr); localStorage.setItem('v_yesterday', yesterdayCount); } else { // 페이지에 접속할 때마다 오늘 방문자 수 1씩 증가 // (단, 새로고침 연타 방지를 위해 세션당 1회 처리가 필요하면 아래 로직 유지) if (!sessionStorage.getItem('visited_session')) { todayCount += 1; totalCount += 1; sessionStorage.setItem('visited_session', 'true'); localStorage.setItem('v_today', todayCount); localStorage.setItem('v_total', totalCount); } } updateVisitorDOM(todayCount, yesterdayCount, totalCount); // 사용자가 페이지를 보고 있을 때 실시간으로 방문자가 늘어나는 생동감 연출 (약 60초마다 랜덤 증가) setInterval(() => { const inc = Math.floor(Math.random() * 2) + 1; // 1~2명씩 증가 todayCount += inc; totalCount += inc; localStorage.setItem('v_today', todayCount); localStorage.setItem('v_total', totalCount); updateVisitorDOM(todayCount, yesterdayCount, totalCount, true); }, 60000); } function updateVisitorDOM(today, yesterday, total, animate = false) { const todayEl = document.getElementById('visitor-today'); const yesterdayEl = document.getElementById('visitor-yesterday'); const totalEl = document.getElementById('visitor-total'); if (todayEl) todayEl.innerText = today.toLocaleString(); if (yesterdayEl) yesterdayEl.innerText = yesterday.toLocaleString(); if (totalEl) totalEl.innerText = total.toLocaleString(); // 숫자가 올라갈 때 살짝 커지는 애니메이션 효과 if (animate && todayEl) { todayEl.classList.add('num-up'); setTimeout(() => todayEl.classList.remove('num-up'), 500); } }
